function proxyBase() {
  return window.location.protocol.startsWith('http') ? window.location.origin : null;
}

function buildSearchQuery(rawText) {
  const cleaned = rawText
    .replace(/^(can you|please|could you|i need to|help me|tell me about|search for|find|look up)\s+/i, '')
    .replace(/[^\w\s?.,'-]/g, ' ')
    .replace(/\s+/g, ' ')
    .trim();
  return cleaned.substring(0, 350);
}

async function scrapeWebPage(url) {
  const base = proxyBase();
  if (!base) return { ok: false, error: 'Local server required for Tavily extraction' };
  
  try {
    const res = await fetch(`${base}/api/proxy?url=${encodeURIComponent(url)}`, { signal: AbortSignal.timeout(25000) });
    if (res.ok) {
      const data = await res.json();
      const text = data.text || data.raw_content || data.content || (Array.isArray(data.results) ? data.results.map(r => r.raw_content || r.content || '').join('\n\n') : '');
      if (text) return { ok: true, text, url };
    }
  } catch (err) {
    console.error("Tavily extract failed:", err);
  }
  return { ok: false, error: 'Failed to extract webpage via Tavily' };
}

async function performLiveWebSearch(query) {
  const cleanQuery = buildSearchQuery(query);
  if (!cleanQuery) return { text: '', results: [] };
  updateSearchStatus(`Searching Tavily for "${cleanQuery.substring(0, 45)}..."`);
  
  const base = proxyBase();
  if (!base) {
    updateSearchStatus("Search failed: Local server required.", true);
    return { text: '', results: [] };
  }

  try {
    const res = await fetch(`${base}/api/proxy?q=${encodeURIComponent(cleanQuery)}&depth=advanced&max_results=8&_t=${Date.now()}`, { 
      method: 'GET',
      headers: { 'Cache-Control': 'no-cache', 'Pragma': 'no-cache' },
      signal: AbortSignal.timeout(20000) 
    });
    if (res.ok) {
      const data = await res.json();
      const rawResults = Array.isArray(data.results) ? data.results : [];
      
      const results = rawResults.map((r, idx) => ({
        index: idx + 1,
        title: r.title || r.url || `Source ${idx + 1}`,
        url: r.url || '',
        content: r.content || r.raw_content || r.snippet || ''
      })).filter(r => r.url || r.content);

      let formattedContext = '';
      if (data.answer) {
        formattedContext += `Direct Answer Summary: ${data.answer}\n\n`;
      }
      if (results.length > 0) {
        formattedContext += results.map(r => 
          `[Source ${r.index}] Title: ${r.title}\nURL: ${r.url}\nSnippet: ${r.content}`
        ).join('\n\n');
      } else if (data.text) {
        formattedContext += data.text;
      }

      updateSearchStatus(`Found ${results.length || (formattedContext ? 1 : 0)} web sources.`, true);
      return { text: formattedContext.trim(), results };
    }
  } catch (e) {
    console.error("Tavily search error:", e);
  }
  
  updateSearchStatus("Search complete.", true);
  return { text: '', results: [] };
}

function renderSourcesHTML(results) {
  if (!results || results.length === 0) return '';
  const valid = results.filter(r => r.url);
  if (valid.length === 0) return '';
  const chips = valid.map((r, idx) => {
    let label = r.title || r.url;
    try { if (!r.title) label = new URL(r.url).hostname.replace(/^www\./, ''); } catch (_) {}
    return `<a class="web-source-chip" href="${escapeHtml(r.url)}" target="_blank" rel="noopener noreferrer" title="${escapeHtml(r.url)}">[${idx + 1}] ${escapeHtml(label)}</a>`;
  }).join('');
  return `<div class="web-sources-box"><div class="web-sources-title">${ICONS.globe} Web Sources</div><div class="web-sources-list">${chips}</div></div>`;
}

function isHttpsPage() { return window.location.protocol === 'https:'; }
function getSameOriginProxy() { return window.location.protocol.startsWith('http') ? `${window.location.origin}/ollama` : null; }

function getCandidateEndpoints() {
  const seen = new Set(); const list = [];
  const add = (u) => {
    if (!u) return;
    const c = u.replace(/\/$/, '');
    if (seen.has(c)) return;
    seen.add(c); list.push(c);
  };
  add(getSameOriginProxy());
  if (!isHttpsPage()) {
    add('http://100.124.111.28:11434');
    if (state.settings.endpoint) add(state.settings.endpoint);
    add('http://localhost:11434');
  }
  return list;
}

function getActiveOllamaBase() {
  const sameOrigin = getSameOriginProxy();
  if (sameOrigin && (isHttpsPage() || state.usingServerProxy)) return sameOrigin;
  const saved = (state.settings.endpoint || '').replace(/\/$/, '');
  if (saved && !(isHttpsPage() && saved.startsWith('http://'))) return saved;
  return sameOrigin || 'http://127.0.0.1:11434';
}

function updateConnStatus(connected, endpoint, reason) {
  const el = document.getElementById('connStatus');
  const text = document.getElementById('connText');
  if (connected && endpoint) {
    state.usingServerProxy = endpoint === getSameOriginProxy();
    el.className = 'connection-status connected';
    text.innerHTML = `Connected: ${endpoint}`;
  } else {
    el.className = 'connection-status disconnected';
    text.textContent = reason || 'Disconnected — check Ollama / network';
  }
}

async function checkConnection() {
  if (!state.authed) {
    updateConnStatus(false, null, 'Sign in to connect');
    return false;
  }
  for (const ep of getCandidateEndpoints()) {
    try {
      const res = await fetch(`${ep}/api/tags`, { method: 'GET', cache: 'no-store' });
      if (res.ok) {
        state.settings.endpoint = ep;
        updateConnStatus(true, ep);
        return true;
      }
    } catch (_) {}
  }
  updateConnStatus(false);
  return false;
}

function shareChat(id, e) {
  e.stopPropagation();
  const chat = state.conversations[id];
  if (!chat) return;

  const safeTitle = (chat.title || 'conversation').replace(/[^a-z0-9]/gi, '_').toLowerCase();
  const filename = `${safeTitle}_chat.json`;
  const jsonStr = JSON.stringify(chat, null, 2);

  const blob = new Blob([jsonStr], { type: 'application/json;charset=utf-8' });
  const url = URL.createObjectURL(blob);

  const a = document.createElement('a');
  a.href = url;
  a.download = filename;
  a.style.display = 'none';
  document.body.appendChild(a);
  a.click();

  setTimeout(() => {
    document.body.removeChild(a);
    URL.revokeObjectURL(url);
  }, 60000);

  showToast('Chat exported successfully', 'success');
}

function handleImportChatFile(event) {
  const file = event.target.files[0];
  if (!file) return;

  const reader = new FileReader();
  reader.onload = function(e) {
    try {
      const importedChat = JSON.parse(e.target.result);
      const newId = 'chat_' + Date.now();
      importedChat.id = newId;
      state.conversations[newId] = importedChat;
      saveConversations();
      renderChatList();
      loadChat(newId);
      showToast(`Imported "${importedChat.title}"`, 'success');
    } catch (err) {
      showToast('Invalid chat file format', 'error');
    }
  };
  reader.readAsText(file);
  event.target.value = '';
}

    }

}

function renderChatList() {
  const container = document.getElementById('chatList');
  const chats = Object.values(state.conversations).sort((a, b) => {
    if (a.pinned !== b.pinned) return b.pinned - a.pinned;
    return new Date(b.createdAt) - new Date(a.createdAt);
  });

  if(chats.length === 0) {
    container.innerHTML = '<div style="padding:20px;text-align:center;color:var(--text-muted);font-size:13px;">No conversations yet</div>';
    return;
  }

  container.innerHTML = chats.map(chat => `
    <div class="chat-item ${chat.id === state.currentChatId ? 'active' : ''}" onclick="loadChat('${chat.id}')">
      <span class="chat-item-icon">${ICONS.chat}</span>
      <span class="chat-item-text">${escapeHtml(chat.title)}</span>
      <div class="chat-item-actions">
        <button class="chat-action-btn pin-btn ${chat.pinned ? 'pinned' : ''}" onclick="togglePinChat('${chat.id}', event)" title="${chat.pinned ? 'Unpin' : 'Pin'}">${ICONS.pin}</button>
        <button class="chat-action-btn" onclick="shareChat('${chat.id}', event)" title="Export/Share Chat">${ICONS.share}</button>

      </div>
    </div>
  `).join('');
}

function formatBytes(b) {
  if (!b || b === 0) return '0 B';
  const k = 1024;
  const sizes = ['B', 'KB', 'MB', 'GB'];
  const i = Math.floor(Math.log(b) / Math.log(k));
  return parseFloat((b / Math.pow(k, i)).toFixed(1)) + ' ' + sizes[i];
}

async function extractTextFromPDF(file) {
  if (!window.pdfjsLib) throw new Error('PDF.js library not loaded');
  const arrayBuffer = await file.arrayBuffer();
  const pdf = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
  let fullText = '';

  for (let pageNum = 1; pageNum <= pdf.numPages; pageNum++) {
    const page = await pdf.getPage(pageNum);
    const textContent = await page.getTextContent();
    let pageText = textContent.items.map(item => item.str).join(' ').trim();

    if (pageText.length < 25 && state.pdfOcrEnabled && window.Tesseract) {
      try {
        const viewport = page.getViewport({ scale: 2.0 });
        const canvas = document.createElement('canvas');
        const ctx = canvas.getContext('2d');
        canvas.width = viewport.width;
        canvas.height = viewport.height;
        await page.render({ canvasContext: ctx, viewport }).promise;
        const ocrRes = await Tesseract.recognize(canvas, 'eng');
        if (ocrRes?.data?.text) pageText = ocrRes.data.text.trim();
      } catch (ocrErr) {
        console.warn(`PDF page ${pageNum} OCR fallback error:`, ocrErr);
      }
    }
    fullText += `--- Page ${pageNum} ---\n${pageText}\n\n`;
  }
  return fullText.trim() || '[PDF contained no extractable text]';
}

async function extractTextFromExcel(file) {
  if (!window.XLSX) throw new Error('SheetJS (XLSX) library not loaded');
  const arrayBuffer = await file.arrayBuffer();
  const workbook = XLSX.read(arrayBuffer, { type: 'array' });
  let result = '';
  workbook.SheetNames.forEach(sheetName => {
    const sheet = workbook.Sheets[sheetName];
    const csv = XLSX.utils.sheet_to_csv(sheet, { blankrows: false });
    if (csv && csv.trim()) {
      result += `--- Sheet: ${sheetName} ---\n${csv.trim()}\n\n`;
    }
  });
  return result.trim() || '[Spreadsheet is empty]';
}

async function extractTextFromDocx(file) {
  const arrayBuffer = await file.arrayBuffer();
  const ext = (file.name.split('.').pop() || '').toLowerCase();

  if (ext === 'docx' && window.mammoth) {
    try {
      const res = await mammoth.extractRawText({ arrayBuffer });
      if (res && res.value && res.value.trim()) return res.value.trim();
    } catch (e) {
      console.warn('Mammoth DOCX parse failed, using fallback:', e);
    }
  }

  const uint8 = new Uint8Array(arrayBuffer);
  const rawLatin = new TextDecoder('latin1').decode(uint8);
  const xmlStripped = rawLatin.replace(/<[^>]+>/g, ' ');
  const blocks = xmlStripped.match(/[\x20-\x7E\r\n\t]{5,}/g) || [];
  const cleaned = blocks.join('\n').replace(/\s{3,}/g, ' ').trim();
  return cleaned || '[Word document contained no extractable text]';
}

async function extractFileContent(file) {
  const name = file.name || '';
  const ext = (name.split('.').pop() || '').toLowerCase();
  const type = (file.type || '').toLowerCase();

  try {
    if (ext === 'pdf' || type === 'application/pdf') {
      return await extractTextFromPDF(file);
    }
    if (['xlsx', 'xls', 'ods'].includes(ext) || type.includes('spreadsheet') || type.includes('excel')) {
      return await extractTextFromExcel(file);
    }
    if (['docx', 'doc', 'rtf'].includes(ext) || type.includes('word')) {
      return await extractTextFromDocx(file);
    }
    if (type.startsWith('image/') || ['png', 'jpg', 'jpeg', 'webp', 'gif', 'bmp'].includes(ext)) {
      return await extractTextFromImage(file);
    }
    if (ext === 'json' || type === 'application/json') {
      const raw = await file.text();
      try {
        return JSON.stringify(JSON.parse(raw), null, 2);
      } catch {
        return raw;
      }
    }
    if (['html', 'htm'].includes(ext) || type === 'text/html') {
      const raw = await file.text();
      const doc = new DOMParser().parseFromString(raw, 'text/html');
      const bodyText = (doc.body?.innerText || '').trim();
      return `[HTML Source]:\n${raw}\n\n[Rendered Text]:\n${bodyText}`;
    }
    return await file.text();
  } catch (err) {
    console.error(`Error extracting ${name}:`, err);
    try {
      return await file.text();
    } catch {
      return `[Failed to extract content from ${name}: ${err.message}]`;
    }
  }
}

function storeDocumentInMemory(file, extractedText) {
  if (!extractedText) return null;

  const existingIdx = state.memoryDocs.findIndex(d => d.name === file.name);
  const docEntry = {
    id: existingIdx >= 0 ? state.memoryDocs[existingIdx].id : ('doc_' + Date.now() + '_' + Math.floor(Math.random() * 1000)),
    name: file.name,
    type: file.type || (file.name.split('.').pop() || 'text/plain'),
    size: file.size,
    content: extractedText,
    enabled: true,
    uploadedAt: new Date().toISOString()
  };

  if (existingIdx >= 0) {
    state.memoryDocs[existingIdx] = docEntry;
  } else {
    if (state.memoryDocs.length >= MAX_MEMORY_DOCS) {
      const removed = state.memoryDocs.shift();
      showToast(`Max ${MAX_MEMORY_DOCS} docs reached. Replaced oldest: ${removed.name}`, 'info');
    }
    state.memoryDocs.push(docEntry);
  }

  saveMemoryDocsToStorage();
  updateMemorySize();
  return docEntry;
}

async function handleFiles(fileList, attachToChat = true) {
  if (!fileList || fileList.length === 0) return;
  const files = Array.from(fileList);

  if (files.length > MAX_MEMORY_DOCS) {
    showToast(`Uploading first ${MAX_MEMORY_DOCS} files (maximum limit is ${MAX_MEMORY_DOCS})`, 'info');
  }
  const slice = files.slice(0, MAX_MEMORY_DOCS);

  updateActivityLine(true);
  for (const file of slice) {
    const isImage = (file.type || '').startsWith('image/');
    if (isImage && !state.photoEnabled) {
      showToast('Photo upload is disabled in Advanced settings', 'error');
      continue;
    }
    if (!isImage && !state.fileEnabled) {
      showToast('File upload is disabled in Advanced settings', 'error');
      continue;
    }

    showToast(`Scanning ${file.name}...`, 'info');
    const extractedText = await extractFileContent(file);
    const savedDoc = storeDocumentInMemory(file, extractedText);

    if (attachToChat) {
      const att = await addPendingAttachment(file, extractedText, savedDoc?.id);
      state.pendingAttachments.push(att);
      renderPendingAttachments();
    }
    showToast(`Saved & indexed "${file.name}"`, 'success');
  }
  updateActivityLine(false);
}

  if (!list) return;
  if (!state.memoryDocs || state.memoryDocs.length === 0) {
    list.innerHTML = '<div style="padding:16px;text-align:center;color:var(--text-muted);font-size:13px;">No documents uploaded yet. Upload up to 10 files (PDF, DOCX, XLSX, PY, JSON, HTML, TXT, CSV).</div>';
    return;
  }
  list.innerHTML = state.memoryDocs.map(doc => {
    const isEnabled = doc.enabled !== false;
    const charCount = doc.content ? doc.content.length : 0;
    return `
      <div class="memory-doc-item" style="opacity:${isEnabled ? '1' : '0.55'};transition:opacity 0.2s;">
        <div style="display:flex;align-items:center;gap:12px;min-width:0;flex:1;">
          <div class="toggle-switch ${isEnabled ? 'active' : ''}" onclick="toggleMemoryDoc('${doc.id}')" title="${isEnabled ? 'Active in chat — click to disable' : 'Disabled — click to enable in chat'}"></div>
          <div style="min-width:0;flex:1;">
            <div style="font-size:13px;color:var(--text-primary);font-weight:600;white-space:nowrap;overflow:hidden;text-overflow:ellipsis;">${escapeHtml(doc.name)}</div>
            <div style="font-size:11px;color:var(--text-muted);margin-top:2px;">${formatBytes(doc.size)} · ${(charCount / 1000).toFixed(1)}k chars extracted · ${isEnabled ? 'Active' : 'Muted'}</div>
          </div>
        </div>
        <div style="display:flex;gap:4px;flex-shrink:0;margin-left:8px;">
          <button class="chat-action-btn" onclick="copyMemoryDoc('${doc.id}')" title="Copy Extracted Text">${ICONS.copy}</button>
          <button class="chat-action-btn chat-item-delete" onclick="deleteMemoryDoc('${doc.id}')" title="Delete Document">${ICONS.trash}</button>
        </div>
      </div>
    `;
  }).join('');
}

function copyMemoryDoc(id) {
  const doc = state.memoryDocs.find(d => d.id === id);
  if (!doc) return;
  navigator.clipboard.writeText(doc.content || '');
  showToast('Copied extracted document text', 'success');
}

function deleteMemoryDoc(id) {

  state.pendingAttachments = state.pendingAttachments.filter(a => a.docId !== id);
  saveMemoryDocsToStorage();

  renderMemoryDocs();
  updateMemorySize();
  showToast('Document removed from knowledge base', 'info');
}

async function sendMessage() {
  const input = document.getElementById('userInput');
  const text = input.value.trim();
  if ((!text && state.pendingAttachments.length === 0) || state.isStreaming) return;

  document.getElementById('welcomeScreen').style.display = 'none';
  if(!state.currentChatId) newChat();

  const currentAttachments = [...state.pendingAttachments];
  state.pendingAttachments = [];
  renderPendingAttachments();
  input.value = '';
  input.style.height = 'auto';

  const chat = state.conversations[state.currentChatId];
  if(chat.messages.length === 0) {
    chat.title = text.substring(0, 50) + (text.length > 50 ? '...' : '');
    renderChatList();
  }

  const attachmentMeta = currentAttachments.map(a => ({ name: a.name, size: a.size, type: a.type }));
  const userMsg = {
    role: 'user',
    content: text || `[Uploaded: ${currentAttachments.map(a => a.name).join(', ')}]`,
    ...(attachmentMeta.length > 0 ? { attachments: attachmentMeta } : {})
  };
  chat.messages.push(userMsg);
  appendMessageWithAttachments('user', userMsg.content, currentAttachments);

  let finalPrompt = text || 'Please analyze the attached document(s).';
  const includedDocNames = new Set();
  const docContextBlocks = [];

  for (const att of currentAttachments) {
    if (att.extractedText && att.extractedText.trim()) {
      includedDocNames.add(att.name);
      docContextBlocks.push(`=== UPLOADED DOCUMENT: ${att.name} ===\n${att.extractedText.substring(0, 250000)}\n=== END OF ${att.name} ===`);
    }
  }

  const activeMemoryDocs = (state.memoryDocs || []).filter(d => d.enabled !== false && d.content && !includedDocNames.has(d.name));
  for (const doc of activeMemoryDocs) {
    includedDocNames.add(doc.name);
    docContextBlocks.push(`=== ACTIVE KNOWLEDGE BASE DOCUMENT: ${doc.name} ===\n${doc.content.substring(0, 250000)}\n=== END OF ${doc.name} ===`);
  }

  if (state.settings.memoryContext && state.settings.memoryContext.trim()) {
    docContextBlocks.push(`=== MEMORY CONTEXT ===\n${state.settings.memoryContext.trim()}\n=== END MEMORY CONTEXT ===`);
  }

  if (docContextBlocks.length > 0) {
    finalPrompt = `${docContextBlocks.join('\n\n')}\n\nUser Query:\n${finalPrompt}`;
  }
  let searchSources = [];
  if (state.isWebConnectivityActive && state.webScrapeEnabled) {
    const urlMatch = text.match(/(https?:\/\/[^\s]+)/);
    if (urlMatch) {
      const targetUrl = urlMatch[0];
      updateSearchStatus(`Scraping ${targetUrl} via Tavily...`);
      const scrapeRes = await scrapeWebPage(targetUrl);
      if (scrapeRes.ok && scrapeRes.text) {
        searchSources = [{ title: targetUrl, url: targetUrl }];
        finalPrompt += `\n\n[Web Page Content from ${targetUrl}]:\n${scrapeRes.text.substring(0, 15000)}\n\nInstruction: Use the scraped content above to answer and include the source URL (${targetUrl}) in your response.`;
        updateSearchStatus("Scrape complete.", true);
      } else {
        updateSearchStatus("Scrape failed.", true);
      }
    } else {
      const searchRes = await performLiveWebSearch(text);
      searchSources = searchRes.results || [];
      if (searchRes.text) {
        finalPrompt += `\n\n[Live Tavily Web Search Results]:\n${searchRes.text}\n\nInstruction: Use the search results above to provide a thorough, accurate answer. You MUST cite your sources inline and include the full https:// URL for every source you reference.`;
      } else {
        updateSearchStatus("Search complete (no results).", true);
      }
    }
  }

  const msgId = 'msg_' + Date.now();
  appendMessageToUI('assistant', '', true, msgId);
  state.isStreaming = true; 
  state.abortController = new AbortController(); 
  updateSendButton(true);

  const msgEl = document.querySelector(`[data-msg-id="${msgId}"] .message-text`);

  let apiMessages = [];
  let systemContent = state.settings.systemPrompt || defaultSettings.systemPrompt;
  if (state.reasoningEnabled) {
    systemContent += '\n\nExtended Reasoning Enabled: Think carefully step-by-step through complex problems before delivering your final answer.';
  }
  if (systemContent) {
    apiMessages.push({ role: 'system', content: systemContent });
  }
  if (state.preserveThinking) {
    const history = chat.messages.slice(0, -1).map(m => ({ role: m.role, content: m.content }));
    apiMessages = apiMessages.concat(history);
  }
  apiMessages.push({ role: 'user', content: finalPrompt });

  try {
    const response = await fetch(getActiveOllamaBase() + '/api/chat', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        model: getCurrentModel(),
        messages: apiMessages,
        think: state.reasoningEnabled,
        stream: true
      }),
      signal: state.abortController.signal
    });

    if(!response.ok) throw new Error(`HTTP ${response.status}`);
    const reader = response.body.getReader();
    const decoder = new TextDecoder();
    let fullContent = '';

    msgEl.innerHTML = '<div class="answer-container streaming-cursor"></div>';
    updateActivityLine(true);
    const answerEl = msgEl.querySelector('.answer-container');

    while(true) {
      const {done, value} = await reader.read();
      if(done) break;
      const chunk = decoder.decode(value, {stream:true});
      const lines = chunk.split('\n').filter(l => l.trim());
      for(const line of lines) {
        try {
          const json = JSON.parse(line);
          if (json.message?.content) {
            fullContent += json.message.content;
            answerEl.innerHTML = formatMessage(fullContent);
            scrollToBottom();
          }
        } catch {}
      }
    }

    answerEl.classList.remove('streaming-cursor');
    if (searchSources.length > 0) {
      msgEl.insertAdjacentHTML('beforeend', renderSourcesHTML(searchSources));
    }
    chat.messages.push({ role: 'assistant', content: fullContent, ...(searchSources.length > 0 ? { sources: searchSources } : {}) });
    saveConversations();
  } catch(err) {
    updateActivityLine(false, err.name !== 'AbortError');
    if (msgEl) msgEl.textContent = 'Error: ' + err.message;
  }

  state.isStreaming = false; 
  updateSendButton(false); 
  saveConversations();
  updateActivityLine(false);
}

function formatMessage(text) {
  if (!text) return '';

  const codeBlocks = [];
  const withPlaceholders = text.replace(/```([a-zA-Z0-9_-]*)\n?([\s\S]*?)```/g, (_, lang, code) => {
    const idx = codeBlocks.length;
    const cleanLang = (lang || 'code').trim();
    const trimmedCode = code.replace(/\n$/, '');
    codeBlocks.push({ lang: cleanLang, code: trimmedCode });
    return `@@CODEBLOCK_${idx}@@`;
  });

  const inlineCodes = [];
  const withInlinePlaceholders = withPlaceholders.replace(/`([^`\n]+)`/g, (_, code) => {
    const idx = inlineCodes.length;
    inlineCodes.push(code);
    return `@@INLINECODE_${idx}@@`;
  });

  const cleaned = withInlinePlaceholders
    .replace(/^#{1,6}\s+/gm, '')       // Removes ## headers at the start of lines
    .replace(/\*\*(.*?)\*\*/g, '$1')   // Removes ** bold markers around words
    .replace(/^\s*[-*_]{2,}\s*$/gm, '') // Removes -- or --- horizontal divider lines
    .replace(/\[([^\]]+)\]\((https?:\/\/[^\s)]+)\)/g, '$1 ($2)'); // Unwraps markdown links to Title (URL)

  const escaped = escapeHtml(cleaned);

  const linked = escaped.replace(
    /(https?:\/\/[^\s<)"']+)/g,
    (url) => {
      const cleanUrl = url.replace(/[.,;!?]+$/, '');
      const trailing = url.slice(cleanUrl.length);
      return `<a href="${cleanUrl}" target="_blank" rel="noopener noreferrer" class="inline-link">${cleanUrl}</a>${trailing}`;
    }
  );

  let html = linked.replace(/\n/g, '<br>');

  html = html.replace(/@@INLINECODE_(\d+)@@/g, (_, idx) => {
    return `<code>${escapeHtml(inlineCodes[Number(idx)])}</code>`;
  });

  html = html.replace(/@@CODEBLOCK_(\d+)@@/g, (_, idx) => {
    const block = codeBlocks[Number(idx)];
    const safeCode = escapeHtml(block.code);
    const safeLang = escapeHtml(block.lang);
    const sendToFindBtn = `<button class="code-copy-btn" onclick="sendCodeBlockToFinder(this, 'find')" title="Send this block to Block Finder (DELETE target)"><svg class="ico ico-sm" viewBox="0 0 24 24"><circle cx="11" cy="11" r="8"/><line x1="21" y1="21" x2="16.65" y2="16.65"/></svg><span>To Find</span></button>`;
    const sendToReplaceBtn = `<button class="code-copy-btn" onclick="sendCodeBlockToFinder(this, 'replace')" title="Send this block to Replacement box (PASTE block)"><svg class="ico ico-sm" viewBox="0 0 24 24"><path d="M16 4h2a2 2 0 0 1 2 2v14a2 2 0 0 1-2 2H6a2 2 0 0 1-2-2V6a2 2 0 0 1 2-2h2"/><rect x="8" y="2" width="8" height="4" rx="1" ry="1"/></svg><span>To Replace</span></button>`;
    const sendToCoderBtn = `<button class="code-copy-btn" onclick="sendCodeBlockToEditor(this)" title="Replace full editor content with this code block"><svg class="ico ico-sm" viewBox="0 0 24 24"><polyline points="16 18 22 12 16 6"/><polyline points="8 6 2 12 8 18"/></svg><span>To Editor</span></button>`;
    const actionBtns = `${sendToFindBtn}${sendToReplaceBtn}${sendToCoderBtn}<button class="code-copy-btn" onclick="copyCodeBlock(this)" title="Copy Code">${copySvgIcon}<span>Copy</span></button>`;
    return `<div class="code-block-wrapper"><div class="code-block-header"><span>${safeLang}</span><div style="display:flex;gap:6px;flex-wrap:wrap;">${actionBtns}</div></div><pre><code>${safeCode}</code></pre><div class="code-block-footer"><div style="display:flex;gap:6px;flex-wrap:wrap;">${actionBtns}</div></div></div>`;
  });

  return html;
}

function escapeHtml(t) {
  const d = document.createElement('div');
  d.textContent = t || '';
  return d.innerHTML;
}

let coderSearchMatches = [];
let coderActiveMatchIdx = -1;
let coderUndoStack = [];

function pushCoderUndo(prevText) {
  if (coderUndoStack.length > 25) coderUndoStack.shift();
  coderUndoStack.push(prevText);
}

function undoCoderEdit() {
  const textarea = document.getElementById('coderTextarea');
  if (!textarea || coderUndoStack.length === 0) {
    showToast('Nothing to undo', 'info');
    return;
  }
  textarea.value = coderUndoStack.pop();
  localStorage.setItem('aiHub_coderContent', textarea.value);
  runCoderSearch(false);
  updateCoderLineNumbers();
  showToast('Reverted last code edit', 'info');
}

function initCoderStudio() {
  state.coderOpen = localStorage.getItem('aiHub_coderOpen') === 'true';
  const savedCode = localStorage.getItem('aiHub_coderContent') || '';
  const savedName = localStorage.getItem('aiHub_coderFilename') || 'index.html';
  const savedWidth = localStorage.getItem('aiHub_coderWidth');

  const panel = document.getElementById('coderPanel');
  const textarea = document.getElementById('coderTextarea');
  const backdrop = document.getElementById('coderBackdrop');
  const filenameInput = document.getElementById('coderFilename');
  const toggleBtn = document.getElementById('coderToggleBtn');
  const findInput = document.getElementById('coderFindInput');

  if (savedWidth && panel) panel.style.width = savedWidth + 'px';
  if (textarea) textarea.value = savedCode;
  if (filenameInput) filenameInput.value = savedName;
  if (state.coderOpen && panel) {
    panel.classList.remove('collapsed');
    toggleBtn?.classList.add('active-coder');
  }
  const toggleBtnLabel = document.getElementById('coderToggleBtnLabel');
  if (toggleBtnLabel) {
    toggleBtnLabel.textContent = state.coderOpen ? 'Close Right Coder Studio' : 'Open Right Coder Studio';
  }
  updateCoderLineNumbers();
  renderCoderBackdrop();
  updateSyntaxHighlighting(); // Initialize syntax highlighting

  textarea?.addEventListener('input', () => {
    runCoderSearch(false);
    updateCoderLineNumbers();
    localStorage.setItem('aiHub_coderContent', textarea.value);
    const status = document.getElementById('coderSavedStatus');
    if (status) status.textContent = 'Saved ' + new Date().toLocaleTimeString();
    updateSyntaxHighlighting(); // Update highlighting on input
  });

  textarea?.addEventListener('scroll', () => {
    const ln = document.getElementById('coderLineNumbers');
    if (ln) ln.scrollTop = textarea.scrollTop;
    if (backdrop) {
      backdrop.scrollTop = textarea.scrollTop;
      backdrop.scrollLeft = textarea.scrollLeft;
    }
  });

  textarea?.addEventListener('keydown', (e) => {
    if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'f') {
      e.preventDefault();
      openCoderFinderWithSelection();
      return;
    }
    if (e.key === 'Tab') {
      e.preventDefault();
      pushCoderUndo(textarea.value);
      const start = textarea.selectionStart;
      const end = textarea.selectionEnd;
      textarea.value = textarea.value.substring(0, start) + '  ' + textarea.value.substring(end);
      textarea.selectionStart = textarea.selectionEnd = start + 2;
      runCoderSearch(false);
      updateCoderLineNumbers();
      localStorage.setItem('aiHub_coderContent', textarea.value);
      updateSyntaxHighlighting();
    }
  });

  findInput?.addEventListener('input', () => {
    runCoderSearch(true);
  });

  findInput?.addEventListener('keydown', (e) => {
    if (e.key === 'Enter' && (e.ctrlKey || e.metaKey)) {
      e.preventDefault();
      stepCoderMatch(e.shiftKey ? -1 : 1);
    }
  });

  filenameInput?.addEventListener('change', () => {
    localStorage.setItem('aiHub_coderFilename', filenameInput.value.trim() || 'index.html');
    updateSyntaxHighlighting(); // Update highlighting when file type changes
  });

  const resizer = document.getElementById('coderResizer');
  let isResizing = false;
  resizer?.addEventListener('mousedown', (e) => {
    isResizing = true;
    resizer.classList.add('dragging');
    panel.style.transition = 'none';
    document.body.style.userSelect = 'none';
    e.preventDefault();
  });

  window.addEventListener('mousemove', (e) => {
    if (!isResizing || !panel) return;
    const newWidth = Math.min(Math.max(window.innerWidth - e.clientX, 340), window.innerWidth * 0.75);
    panel.style.width = newWidth + 'px';
    localStorage.setItem('aiHub_coderWidth', Math.round(newWidth));
  });

  window.addEventListener('mouseup', () => {
    if (isResizing && panel) {
      isResizing = false;
      resizer?.classList.remove('dragging');
      panel.style.transition = '';
      document.body.style.userSelect = '';
    }
  });
}

function toggleCoderFinder(forceState) {
  const drawer = document.getElementById('coderFinderDrawer');
  const btn = document.getElementById('coderFinderToggleBtn');
  if (!drawer) return;
  const shouldOpen = typeof forceState === 'boolean' ? forceState : !drawer.classList.contains('open');
  drawer.classList.toggle('open', shouldOpen);
  btn?.classList.toggle('active-finder', shouldOpen);
  if (shouldOpen) {
    document.getElementById('coderFindInput')?.focus();
    runCoderSearch(true);
  }
}

function findCodeMatches(code, query, caseSensitive, smartIndent) {
  if (!query) return [];
  const matches = [];
  const normQuery = query.replace(/\r\n/g, '\n');

  // 1. Exact or case-insensitive direct match
  const flags = caseSensitive ? 'g' : 'gi';
  const exactRegex = new RegExp(escapeRegExp(normQuery), flags);
  let m;
  while ((m = exactRegex.exec(code)) !== null) {
    if (m[0].length === 0) { exactRegex.lastIndex++; continue; }
    matches.push({ start: m.index, end: m.index + m[0].length });
  }

  // 2. Smart multi-line indentation-tolerant fallback if exact match didn't hit
  if (matches.length === 0 && smartIndent && normQuery.includes('\n')) {
    const lines = normQuery.split('\n').map(l => l.trim());
    while (lines.length > 0 && lines[0] === '') lines.shift();
    while (lines.length > 0 && lines[lines.length - 1] === '') lines.pop();
    if (lines.length > 0) {
      const pattern = lines.map(l => l === '' ? '[ \\t]*' : '[ \\t]*' + escapeRegExp(l) + '[ \\t]*').join('\\r?\\n');
      try {
        const flexRegex = new RegExp(pattern, flags);
        while ((m = flexRegex.exec(code)) !== null) {
          if (m[0].length === 0) { flexRegex.lastIndex++; continue; }
          matches.push({ start: m.index, end: m.index + m[0].length });
        }
      } catch (_) {}
    }
  }
  return matches;
}

function runCoderSearch(scrollToFirst = true) {
  const textarea = document.getElementById('coderTextarea');
  const findInput = document.getElementById('coderFindInput');
  if (!textarea || !findInput) return;

  const code = textarea.value || '';
  const query = findInput.value || '';
  const caseSensitive = !!document.getElementById('coderCaseSensitive')?.checked;
  const smartIndent = !!document.getElementById('coderSmartIndent')?.checked;

  if (!query) {
    coderSearchMatches = [];
    coderActiveMatchIdx = -1;
    updateMatchBadge();
    renderCoderBackdrop();
    updateCoderLineNumbers();
    return;
  }

  coderSearchMatches = findCodeMatches(code, query, caseSensitive, smartIndent);
  if (coderSearchMatches.length === 0) {
    coderActiveMatchIdx = -1;
  } else if (coderActiveMatchIdx < 0 || coderActiveMatchIdx >= coderSearchMatches.length) {
    coderActiveMatchIdx = 0;
  }

  updateMatchBadge();
  renderCoderBackdrop();
  updateCoderLineNumbers();
  if (scrollToFirst && coderActiveMatchIdx >= 0) {
    scrollToCoderMatch(coderActiveMatchIdx);
  }
}

function highlightCodeSegment(rawSegment, language) {
  if (!rawSegment) return '';
  const escaped = escapeHtml(rawSegment);

  if (language === 'html') {
    return escaped.replace(
      /(<!--[\s\S]*?-->)|(<\/?[a-zA-Z0-9\-]+)|(\s[a-zA-Z\-:]+=)("[^"]*"|'[^']*')|(>)/g,
      (match, comment, tag, attr, str, closeTag) => {
        if (comment) return `${comment}`;
        if (tag) return `${tag}`;
        if (attr) return `${attr}${str}`;
        if (closeTag) return `${closeTag}`;
        return match;
      }
    );
  }
  if (language === 'css') {
    return escaped.replace(
      /(\/\*[\s\S]*?\*\/)|([.#][a-zA-Z0-9\-_]+)|(\{|\})|([a-zA-Z\-]+)(\s*:\s*)([^;{}\n]+)/g,
      (match, comment, selector, bracket, prop, colon, val) => {
        if (comment) return `${comment}`;
        if (selector) return `${selector}`;
        if (bracket) return `${bracket}`;
        if (prop) return `${prop}${colon}${val}`;
        return match;
      }
    );
  }
  if (language === 'javascript') {
    return escaped.replace(
      /(\/\/[^\n]*|\/\*[\s\S]*?\*\/)|('[^'\n]*'|"[^"\n]*"|`[\s\S]*?`)|\b(function|return|if|else|for|while|do|switch|case|break|continue|try|catch|finally|throw|var|let|const|class|extends|import|export|default|from|as|async|await|new|typeof|instanceof)\b|\b(true|false|null|undefined|NaN|Infinity)\b|\b([0-9]+(?:\.[0-9]+)?)\b|(\.[a-zA-Z_$][a-zA-Z0-9_$]*)/g,
      (match, comment, str, kw, atom, num, prop) => {
        if (comment) return `${comment}`;
        if (str) return `${str}`;
        if (kw) return `${kw}`;
        if (atom) return `${atom}`;
        if (num) return `${num}`;
        if (prop) return `${prop}`;
        return match;
      }
    );
  }
  if (language === 'python') {
    return escaped.replace(
      /(#[^\n]*)|('''[\s\S]*?'''|"""[\s\S]*?"""|'[^'\n]*'|"[^"\n]*")|\b(def|class|if|elif|else|for|while|try|except|finally|with|import|from|as|return|yield|lambda|and|or|not|in|is|assert|break|continue|del|global|nonlocal|pass| print|raise|async|await)\b|\b(True|False|None)\b|\b([0-9]+(?:\.[0-9]+)?)\b/g,
      (match, comment, str, kw, atom, num) => {
        if (comment) return `${comment}`;
        if (str) return `${str}`;
        if (kw) return `${kw}`;
        if (atom) return `${atom}`;
        if (num) return `${num}`;
        return match;
      }
    );
  }
  if (language === 'json') {
    return escaped.replace(
      /("[^"\n]*"\s*:)|("[^"\n]*")|\b(true|false|null)\b|\b(-?[0-9]+(?:\.[0-9]+)?)\b/g,
      (match, key, str, atom, num) => {
        if (key) return `${key}`;
        if (str) return `${str}`;
        if (atom) return `${atom}`;
        if (num) return `${num}`;
        return match;
      }
    );
  }
  return escaped;
}

function updateSyntaxHighlighting() {
  const textarea = document.getElementById('coderTextarea');
  const backdrop = document.getElementById('coderBackdrop');
  const filename = document.getElementById('coderFilename')?.value || '';
  if (!textarea || !backdrop) return;

  const code = textarea.value || '';
  const extension = filename.split('.').pop()?.toLowerCase();

  let language = 'plaintext';
  if (extension === 'html' || extension === 'htm') language = 'html';
  else if (extension === 'css') language = 'css';
  else if (extension === 'js' || extension === 'jsx' || extension === 'ts' || extension === 'tsx') language = 'javascript';
  else if (extension === 'py') language = 'python';
  else if (extension === 'json') language = 'json';

  if (coderSearchMatches.length === 0) {
    backdrop.innerHTML = highlightCodeSegment(code, language) + '\n';
  } else {
    let finalHtml = '';
    let lastIndex = 0;
    coderSearchMatches.forEach((m, idx) => {
      finalHtml += highlightCodeSegment(code.slice(lastIndex, m.start), language);
      const cls = idx === coderActiveMatchIdx ? 'code-find-match code-find-active' : 'code-find-match';
      finalHtml += `${highlightCodeSegment(code.slice(m.start, m.end), language)}`;
      lastIndex = m.end;
    });
    finalHtml += highlightCodeSegment(code.slice(lastIndex), language) + '\n';
    backdrop.innerHTML = finalHtml;
  }

  backdrop.scrollTop = textarea.scrollTop;
  backdrop.scrollLeft = textarea.scrollLeft;
}
